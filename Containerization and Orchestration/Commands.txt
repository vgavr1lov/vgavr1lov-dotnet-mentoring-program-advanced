$response = curl.exe -sk -X POST http://127.0.0.1:5000/connect/token `
  -H "Content-Type: application/x-www-form-urlencoded" `
  --data-urlencode "grant_type=password" `
  --data-urlencode "username=manager" `
  --data-urlencode "password=manager" `
  --data-urlencode "scope=openid profile roles catalog.read catalog.create catalog.update catalog.delete offline_access" `
  --data-urlencode "client_id=postman" `
  --data-urlencode "client_secret=postman" | ConvertFrom-Json

$token = $response.access_token

curl.exe -i -X PUT http://127.0.0.1:5001/v1/products `
  -H "Content-Type: application/json" `
  -H "Authorization: Bearer $token" `
  -d '{\"id\": 1, \"name\": \"Laptop - updated7676\", \"description\": \"Laptop - updated\", \"imageUrl\": \"https://example.com/laptop.jpg\", \"imageAltText\": \"Laptop - updated\", \"categoryId\": 1, \"amount\": 120, \"currency\": \"PLN\", \"quantity\": 1}'

curl.exe -i -X POST http://127.0.0.1:5002/v1/carts/1/items `
  -H "Content-Type: application/json" `
  -H "Authorization: Bearer $token" `
  -d '{\"id\": 1, \"name\": \"product\", \"imageUrl\": \"https://example.com/product.jpg\", \"imageAltText\": \"product\", \"amount\": 10, \"currency\": \"PLN\", \"quantity\": 1}'

curl.exe -i http://127.0.0.1:5002/v1/carts/1 `
